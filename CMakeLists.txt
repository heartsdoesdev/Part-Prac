cmake_minimum_required(VERSION 3.25)
project(PartPrac VERSION 1.0.4 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(${PROJECT_NAME} SHARED src/main.cpp)
if(NOT DEFINED ENV{GEODE_SDK})
    message(FATAL_ERROR "Set GEODE_SDK to the Geode 5.11.1 SDK folder")
endif()
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
setup_geode_mod(${PROJECT_NAME} DONT_INSTALL)
enable_testing()
add_executable(range_tests tests/range_tests.cpp)
add_test(NAME range_tests COMMAND range_tests)
add_executable(scene_tests tests/scene_tests.cpp)
add_test(NAME scene_tests COMMAND scene_tests)
add_executable(start_position_tests tests/start_position_tests.cpp)
add_test(NAME start_position_tests COMMAND start_position_tests)

add_executable(exit_tests tests/exit_tests.cpp)
add_test(NAME exit_tests COMMAND exit_tests)
